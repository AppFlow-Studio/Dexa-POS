begin;

drop policy if exists "device_config_history_select_hq" on public.device_config_history;
drop policy if exists "device_config_history_select_merchant" on public.device_config_history;
drop policy if exists "device_config_history_select_carrier" on public.device_config_history;

create policy "device_config_history_select_combined"
on public.device_config_history
for select
to authenticated
using (
  is_dexapos_admin()
  or exists (
    select 1
    from device_inventory di
    where di.id = device_config_history.device_id
      and di.merchant_id is not null
      and is_merchant_admin(di.merchant_id)
  )
  or exists (
    select 1
    from device_inventory di
    join merchants m on m.id = di.merchant_id
    where di.id = device_config_history.device_id
      and m.carrier_id = get_my_carrier_id()
  )
);

commit;

ON public.device_inventory
FOR SELECT
TO authenticated
USING (
  is_dexapos_admin()
  OR (
    merchant_id IS NOT NULL
    AND is_merchant_admin(merchant_id)
  )
  OR (
    merchant_id IS NOT NULL
    AND EXISTS (
      SELECT 1
      FROM public.merchants m
      WHERE m.id = device_inventory.merchant_id
        AND m.carrier_id = get_my_carrier_id()
    )
  )
);

DROP POLICY IF EXISTS "lbp_carrier_admin_read" ON public.location_banking_profiles;
DROP POLICY IF EXISTS "lbp_hq_admin_all" ON public.location_banking_profiles;
DROP POLICY IF EXISTS "lbp_merchant_owner_rw" ON public.location_banking_profiles;

CREATE POLICY "lbp_hq_admin_all"
ON public.location_banking_profiles
FOR ALL
TO authenticated
USING (is_dexapos_admin())
WITH CHECK (is_dexapos_admin());

CREATE POLICY "lbp_merchant_owner_rw"
ON public.location_banking_profiles
FOR ALL
TO authenticated
USING (
  EXISTS (
    SELECT 1
    FROM public.locations l
    JOIN public.merchants mer ON mer.id = l.merchant_id
    JOIN public.members mm ON mm.organization_id = mer.clerk_org_id
    WHERE l.id = location_banking_profiles.location_id
      AND mm.user_id = current_user_id()
      AND mm.role = 'merchant.owner'
  )
)
WITH CHECK (
  EXISTS (
    SELECT 1
    FROM public.locations l
    JOIN public.merchants mer ON mer.id = l.merchant_id
    JOIN public.members mm ON mm.organization_id = mer.clerk_org_id
    WHERE l.id = location_banking_profiles.location_id
      AND mm.user_id = current_user_id()
      AND mm.role = 'merchant.owner'
  )
);

CREATE POLICY "lbp_carrier_admin_read"
ON public.location_banking_profiles
FOR SELECT
TO authenticated
USING (
  EXISTS (
    SELECT 1
    FROM public.locations l
    JOIN public.merchants mer ON mer.id = l.merchant_id
    JOIN public.carriers c ON c.id = mer.carrier_id
    JOIN public.members cm ON cm.organization_id = c.clerk_org_id
    JOIN public.roles cr ON cr.code = cm.role
    WHERE l.id = location_banking_profiles.location_id
      AND cm.user_id = current_user_id()
      AND cr.organization_type = 'carrier'::public.organization_type
  )
);

on public.support_tickets
as permissive
for all
to public
using (
  (
    exists (
      select 1
      from members mem
      join organizations o on o.id = mem.organization_id
      where mem.user_id = (auth.uid())::text
        and o.id = (
          select organizations.id
          from organizations
          where organizations.id = current_setting('app.dexa_hq_org_id', true)
          limit 1
        )
    )
  )
  or
  (
    merchant_id in (
      select m.id
      from merchants m
      where m.clerk_org_id = (
        select organizations.id
        from organizations
        where organizations.id = ((auth.jwt() -> 'org') ->> 'id')
        limit 1
      )
    )
  )
)
with check (
  (
    exists (
      select 1
      from members mem
      join organizations o on o.id = mem.organization_id
      where mem.user_id = (auth.uid())::text
        and o.id = (
          select organizations.id
          from organizations
          where organizations.id = current_setting('app.dexa_hq_org_id', true)
          limit 1
        )
    )
  )
  or

      where m.clerk_org_id = (
