do $$
declare
  p record;
  new_qual text;
  new_with_check text;
  roles_clause text;
  create_sql text;
begin
  for p in
    select *
    from pg_policies
    where schemaname = 'public'
      and tablename = 'staff_profiles'
      and (
        coalesce(qual, '') like '%auth.uid()%' or
        coalesce(with_check, '') like '%auth.uid()%'
      )
  loop
    new_qual := case when p.qual is null then null else replace(p.qual, 'auth.uid()', '(select auth.uid())') end;
    new_with_check := case when p.with_check is null then null else replace(p.with_check, 'auth.uid()', '(select auth.uid())') end;

    select coalesce(string_agg(quote_ident(r), ', '), 'public')
      into roles_clause
    from unnest(p.roles) as r;

    execute format('drop policy if exists %I on %I.%I;', p.policyname, p.schemaname, p.tablename);

    create_sql := format(
      'create policy %I on %I.%I as %s for %s to %s',
      p.policyname,
      p.schemaname,
      p.tablename,
      p.permissive,
      p.cmd,
      roles_clause
    );

    if new_qual is not null then
      create_sql := create_sql || format(' using (%s)', new_qual);
    end if;

    if new_with_check is not null then
      create_sql := create_sql || format(' with check (%s)', new_with_check);
    end if;

    create_sql := create_sql || ';';
    execute create_sql;
  end loop;
end
$$;
